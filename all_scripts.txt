INDEX_FILE = (
    EXTRACTED /

NUMBERS_FILE = (

# الكلمات التي تقل ثقتها عن هذا الرقم توضع للمراجعة فقط
LOW_CONFIDENCE_THRESHOLD = 70


# =====================================================
# قراءة البيانات
# =====================================================

def load_json(path):

    with open(path, encoding="utf-8") as f:
        return json.load(f)


def clean_text(text):

    return re.sub(r"\s+", " ", text).strip()


# =====================================================
# استخراج الكلمات من سطر OCR
# =====================================================

def extract_line(line_element):

    words = []

    word_elements = line_element.find_all(
        "span",
        class_="ocrx_word"
    )

    for element in word_elements:

        word = clean_text(element.get_text("", strip=True))

        if not word:
            continue

        title = element.get("title", "")

        confidence_match = re.search(
            r"x_wconf\s+(\d+)",
            title
        )

        confidence = (
            int(confidence_match.group(1))
            if confidence_match
            else None
        )

        words.append({
            "text": word,
            "confidence": confidence
        })

    return words


# =====================================================
# استخراج صفحة واحدة
# =====================================================

def extract_page(page_number, record, hocr_bytes, page_number_data):

    text_start, text_end, byte_start, byte_end = record

    # التحقق من حدود البيانات
    if not (0 <= byte_start < byte_end <= len(hocr_bytes)):
        raise ValueError(
            f"حدود غير صحيحة في الصفحة {page_number}"
        )

    # استخراج مقطع الصفحة باستخدام إزاحات البايتات
    page_bytes = hocr_bytes[byte_start:byte_end]

    page_html = page_bytes.decode(
        "utf-8",
        errors="replace"
    )

    soup = BeautifulSoup(
        page_html,
        "html.parser"
    )

    page_element = soup.find(
        "div",
        class_="ocr_page"
    )

    if page_element is None:

        raise ValueError(
            f"لم يتم العثور على عنصر ocr_page في الصفحة {page_number}"
        )

    page_id = page_element.get("id")

    # معلومات الصفحة المطبوعة
    printed_number = page_number_data.get(
        page_number,
        {}
    ).get("pageNumber", "")

    # استخراج الأسطر
    line_elements = page_element.find_all(
        class_="ocr_line"
    )

    lines = []
    all_words = []

    for line_index, line_element in enumerate(line_elements):

        words = extract_line(line_element)

        if not words:
            continue

        line_text = " ".join(
            word["text"] for word in words
        )

        lines.append({
            "line_number": line_index,
            "text": line_text,
            "words": words
        })

        all_words.extend(words)

    # احتياط: إذا لم توجد وسوم ocr_line
    if not lines:

        words = extract_line(page_element)

        if words:

            all_words = words

            lines.append({
                "line_number": 0,
                "text": " ".join(
                    word["text"] for word in words
                ),
                "words": words
            })

    page_text = "\n".join(
        line["text"] for line in lines
    )

    # حساب الثقة
    confidences = [
        word["confidence"]
        for word in all_words
        if word["confidence"] is not None
    ]

    average_confidence = (
        round(statistics.mean(confidences), 2)
        if confidences
        else None
    )

    low_confidence_words = [
        word
        for word in all_words
        if (
            word["confidence"] is not None
            and word["confidence"] < LOW_CONFIDENCE_THRESHOLD
        )
    ]

    result = {
        "leaf_number": page_number,

        "page_id": page_id,

        "printed_page_number": printed_number,

        "source": {
            "file": HOCR_FILE.name,
            "byte_start": byte_start,
            "byte_end": byte_end,
            "text_index_start": text_start,
            "text_index_end": text_end
        },

        "statistics": {
            "line_count": len(lines),
            "word_count": len(all_words),
            "average_word_confidence": average_confidence,
            "low_confidence_word_count": len(
                low_confidence_words
            )
        },

        "text": page_text,

        "lines": lines,

        "low_confidence_words": low_confidence_words
    }

    return result

def main():

        print(
            "الفرق:",

            "لم يتم العثور على بداية صفحة HOCR في الإزاحة الأولى"
        )

    print("\n[3] استخراج الصفحات...")

    txt_path = EXTRACTED / "pages.txt"

    report = {
        "total_pages_in_index": len(index),
        "successful_pages": 0,
        "failed_pages": [],
        "total_words": 0,
        "total_lines": 0,
        "pages_with_printed_numbers": 0,
        "pages_with_low_confidence_words": 0
    }

        open(jsonl_path, "w", encoding="utf-8") as jsonl,

            try:

                report["total_words"] += stats["word_count"]

        report_path,

            f,
            ensure_ascii=False,
            indent=2
        )

    print("اكتمل استخراج الصفحات")
    print("=" * 65)

    print(
        "الصفحات الناجحة:",

    )

        "الصفحات الفاشلة:",
        len(report["failed_pages"])
    )

        "إجمالي الكلمات:",

        "إجمالي الأسطر:",

    )

    )

    print(PAGES_DIR)
    print(jsonl_path)
    print(txt_path)
    print(report_path)


if __name__ == "__main__":

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import json
from pathlib import Path

BASE = Path("volume_01/extracted")

PAGES_DIR = BASE / "pages"
OUTPUT_DIR = BASE / "pages_enriched"

INDEX_FILE = BASE / "معجم البلدان والقبائل اليمنية المجلد 1_hocr_pageindex.json"
SEARCH_FILE = BASE / "معجم البلدان والقبائل اليمنية المجلد 1_hocr_searchtext.txt"

JSONL_OUTPUT = BASE / "pages_enriched.jsonl"
TEXT_OUTPUT = BASE / "pages_enriched.txt"

def load_json(path):
    return json.loads(path.read_text(encoding="utf-8"))

def main():

    print("=" * 65)
    print("ربط صفحات HOCR بنص Search Text")
    print("=" * 65)

    index = load_json(INDEX_FILE)

    # قراءة النص كاملاً دون تغيير أو تطبيع
    search_text = SEARCH_FILE.read_text(encoding="utf-8")

    if len(index) != 828:
        raise ValueError(f"عدد سجلات الفهرس غير متوقع: {len(index)}")

    if len(index) != len(list(PAGES_DIR.glob("page_*.json"))):
        raise ValueError("عدد ملفات الصفحات لا يطابق عدد سجلات الفهرس")

    print("عدد الصفحات:", len(index))
    print("عدد محارف Search Text:", len(search_text))

    # التحقق من اتصال الحدود وتسلسلها
    if index[0][0] != 0:
        raise ValueError("بداية الفهرس لا تساوي صفر")

    for i, record in enumerate(index):
        start, end = record[0], record[1]

        if not (0 <= start <= end <= len(search_text)):
            raise ValueError(f"حدود غير صحيحة في الصفحة {i}")

        if i > 0 and start != index[i - 1][1]:
            raise ValueError(f"فجوة أو تداخل قبل الصفحة {i}")

    if index[-1][1] != len(search_text):
        raise ValueError("نهاية الفهرس لا تطابق طول Search Text")

    print("التحقق من الحدود: ناجح")

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    empty_pages = []
    total_chars = 0

    with JSONL_OUTPUT.open("w", encoding="utf-8") as jsonl, \
         TEXT_OUTPUT.open("w", encoding="utf-8") as txt:

        for page_no, record in enumerate(index):

            start, end = record[0], record[1]

            source_file = PAGES_DIR / f"page_{page_no:04d}.json"

            if not source_file.exists():
                raise FileNotFoundError(source_file)

            page = load_json(source_file)

            if page["leaf_number"] != page_no:
                raise ValueError(f"رقم الصفحة غير مطابق: {page_no}")

            # استخراج النص الموازي وفق إزاحات المحارف
            parallel_text = search_text[start:end]

            page["parallel_text"] = {
                "source_file": SEARCH_FILE.name,
                "offset_unit": "unicode_characters",
                "start": start,
                "end": end,
                "character_count": len(parallel_text),
                "text": parallel_text
            }

            output_file = OUTPUT_DIR / source_file.name

            output_file.write_text(
                json.dumps(page, ensure_ascii=False, indent=2),
                encoding="utf-8"
            )

            jsonl.write(
                json.dumps(page, ensure_ascii=False) + "\n"
            )

            txt.write(
                f"\n{'=' * 70}\n"
                f"رقم الورقة: {page_no}\n"
                f"HOCR: {page['page_id']}\n"
                f"{'=' * 70}\n\n"
                f"[HOCR]\n{page['text']}\n\n"
                f"[SEARCH TEXT]\n{parallel_text}\n"
            )

            total_chars += len(parallel_text)

            if not parallel_text.strip():
                empty_pages.append(page_no)

    print("\n" + "=" * 65)
    print("اكتمل الربط بنجاح")
    print("=" * 65)

    print("الصفحات المعالجة:", len(index))
    print("إجمالي المحارف المرتبطة:", total_chars)
    print("الصفحات الفارغة:", len(empty_pages))
    print("أرقام الصفحات الفارغة:", empty_pages)

    print("\nالمخرجات:")
    print("JSON:", OUTPUT_DIR)
    print("JSONL:", JSONL_OUTPUT)
    print("TXT:", TEXT_OUTPUT)

if __name__ == "__main__":
    main()

import json
from pathlib import Path
from difflib import SequenceMatcher

PAGES = Path("volume_01/extracted/pages_enriched")

samples = [0, 1, 4, 10, 100, 400, 600, 824, 825, 826, 827]

for number in samples:

    path = PAGES / f"page_{number:04d}.json"

    if not path.exists():
        continue

    page = json.loads(path.read_text(encoding="utf-8"))

    hocr = page["text"]
    search = page["parallel_text"]["text"]

    matcher = SequenceMatcher(None, hocr, search)

    print("\n" + "=" * 70)
    print("الورقة:", number)
    print("HOCR:", len(hocr), "محرف")
    print("Search Text:", len(search), "محرف")
    print("نسبة التشابه التقريبية:", round(matcher.ratio() * 100, 2), "%")

    print("\n--- HOCR ---")
    print(hocr[:1000])

    print("\n--- SEARCH TEXT ---")
    print(search[:1000])

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Convert enriched HOCR pages into OpenITI mARkdown.

Input:
    volume_01/extracted/pages_enriched/page_XXXX.json

Output:
    volume_01/openiti_output/
        *.mARkdown
        *.yml
        page_map.csv
        conversion_report.json

Important:
    - HOCR remains the primary text.
    - Search Text is NOT merged into HOCR.
    - No OCR correction or normalization is performed.
    - Empty pages are preserved.
    - Page markers represent sequential scanned leaves,
      NOT verified printed page numbers.
"""

import json
import csv
import re
import sys

from pathlib import Path
from datetime import date


# ============================================================
# 1. SETTINGS
# ============================================================

BASE_DIR = Path(__file__).resolve().parent

SOURCE_DIR = (
    BASE_DIR
    / "volume_01"
    / "extracted"
    / "pages_enriched"
)

OUTPUT_DIR = (
    BASE_DIR
    / "volume_01"
    / "openiti_output"
)

# بيانات تعريف مؤقتة.
# يجب تعديل اسم المؤلف ومعرّف OpenITI عند التحقق منهما.

BOOK_TITLE = "معجم البلدان والقبائل اليمنية"

AUTHOR_AR = "غير متحقق منه"

# URI مؤقت، وليس معرّف OpenITI رسميّاً معتمداً.
URI = (
    "0000Unknown."
    "MujamBuldanQabailYemen."
    "IA20200711-ara1"
)

ARCHIVE_ID = "2_20200711_20200711_1634"

ARCHIVE_URL = (
    "https://archive.org/details/"
    + ARCHIVE_ID
)

VOLUME_NUMBER = 1

EXPECTED_PAGES = 828

# ============================================================
# 2. HELPERS
# ============================================================


def load_json(path):
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)


def write_text(path, content):
    with path.open("w", encoding="utf-8", newline="\n") as f:
        f.write(content)


def count_words(text):
    return len(re.findall(r"\S+", text))


def get_page_files():
    if not SOURCE_DIR.exists():
        raise FileNotFoundError(
            f"مجلد المصدر غير موجود:\n{SOURCE_DIR}"
        )

    files = list(SOURCE_DIR.glob("page_*.json"))

    def page_number(path):
        match = re.fullmatch(
            r"page_(\d+)\.json",
            path.name
        )

        if not match:
            raise ValueError(
                f"اسم ملف غير صالح: {path.name}"
            )

        return int(match.group(1))

    return sorted(files, key=page_number)


# ============================================================
# 3. VALIDATE INPUT
# ============================================================


def validate_pages(files):

    if len(files) != EXPECTED_PAGES:
        raise ValueError(
            f"عدد ملفات الصفحات غير صحيح: "
            f"{len(files)}، المتوقع {EXPECTED_PAGES}"
        )

    pages = []

    for expected_index, path in enumerate(files):

        page = load_json(path)

        leaf = page.get("leaf_number")

        if leaf != expected_index:
            raise ValueError(
                f"اختلاف ترتيب الصفحات: {path.name}\n"
                f"leaf_number = {leaf}\n"
                f"المتوقع = {expected_index}"
            )

        if not isinstance(page.get("text"), str):
            raise ValueError(
                f"حقل text غير صالح في {path.name}"
            )

        if not isinstance(page.get("lines"), list):
            raise ValueError(
                f"حقل lines غير صالح في {path.name}"
            )

        pages.append(page)

    return pages


# ============================================================
# 4. CONVERT ONE PAGE TO OPENITI
# ============================================================


def convert_page_text(text):

    """
    Convert HOCR page text into OpenITI paragraph lines.

    Each original OCR line becomes an OpenITI paragraph.

    No words are corrected, reordered, or normalized.
    """

    output = []

    # فصل الأسطر دون تنظيف النص.
    original_lines = text.split("\n")

    for line in original_lines:

        # إزالة CR فقط إذا كان جزءاً من نهاية CRLF.
        if line.endswith("\r"):
            line = line[:-1]

        if line == "":
            continue

        # OpenITI paragraph marker.
        output.append("# " + line)

    return output


# ============================================================
# 5. BUILD OPENITI DOCUMENT
# ============================================================


def build_document(pages):

    output = []

    # OpenITI magic value.
    output.append("######OpenITI#")

    # Non-machine-readable metadata.
    output.append(
        f"#META#Title: {BOOK_TITLE}"
    )

    output.append(
        f"#META#Author: {AUTHOR_AR}"
    )

    output.append(
        "#META#Language: Arabic"
    )

    output.append(
        "#META#Source: Internet Archive"
    )

    output.append(
        f"#META#Source ID: {ARCHIVE_ID}"
    )

    output.append(
        f"#META#Source URL: {ARCHIVE_URL}"
    )

    output.append(
        "#META#Primary OCR: HOCR"
    )

    output.append(
        "#META#Parallel OCR: Search Text, retained separately"
    )

    output.append(
        "#META#Pagination: Sequential scanned leaves, "
        "not verified printed pagination"
    )

    output.append(
        "#META#Conversion: Automatic OCR-to-OpenITI conversion"
    )

    output.append(
        "#META#Structural annotation: Not performed"
    )

    # End of metadata header.
    output.append("#META#Header#End#")

    total_characters = 0
    total_words = 0
    empty_pages = []

    page_map = []

    # --------------------------------------------------------
    # Process pages in original order.
    # --------------------------------------------------------

    for index, page in enumerate(pages):

        text = page["text"]

        total_characters += len(text)
        total_words += count_words(text)

        if not text:
            empty_pages.append(index)

        # Convert text lines.
        converted_lines = convert_page_text(text)

        output.extend(converted_lines)

        # Sequential OpenITI page marker.
        page_number = index + 1

        page_marker = (
            f"PageV{VOLUME_NUMBER:02d}"
            f"P{page_number:03d}"
        )

        output.append(page_marker)

        # Keep page boundaries visible.
        output.append("")

        # Record original page information.
        page_map.append({
            "leaf_number": index,
            "openiti_page": page_marker,
            "printed_page_number": (
                page.get("printed_page_number") or ""
            ),
            "hocr_characters": len(text),
            "hocr_words": count_words(text),
            "is_empty": not bool(text),
        })

    document = "\n".join(output)

    # Ensure a final newline.
    document += "\n"

    statistics = {
        "pages": len(pages),
        "characters": total_characters,
        "words": total_words,
        "empty_pages": empty_pages,
        "empty_page_count": len(empty_pages),
    }

    return document, page_map, statistics


# ============================================================
# 6. GENERATE VERSION YAML
# ============================================================


def build_yaml(statistics):

    today = date.today().isoformat()

    lines = [
        f"00#VERS#CLENGTH##: {statistics['characters']}",
        f"00#VERS#LENGTH###: {statistics['words']}",
        f"00#VERS#URI######: {URI}",
        f"80#VERS#LINKS####: {ARCHIVE_URL}",
        f"90#VERS#DATE#####: {today}",
        (
            "90#VERS#COMMENT##: "
            "Automatically converted from Internet Archive HOCR OCR. "
            "The primary text is preserved without OCR correction. "
            "Search Text remains a separate parallel transcription. "

def verify_output(markdown_path, pages):

    content = markdown_path.read_text(
        encoding="utf-8"
    )

    lines = content.splitlines()

    if not lines:
        raise ValueError("ملف OpenITI فارغ")

    if lines[0] != "######OpenITI#":
        raise ValueError("قيمة OpenITI magic غير صحيحة")

    if "#META#Header#End#" not in lines:
        raise ValueError("فاصل البيانات الوصفية مفقود")

    markers = re.findall(
        r"^PageV01P\d{3}$",
        content,
        flags=re.MULTILINE
    )

    if len(markers) != len(pages):
        raise ValueError(
            f"عدد وسوم الصفحات غير صحيح: {len(markers)}"
        )

    expected_markers = [
        f"PageV01P{i:03d}"
        for i in range(1, len(pages) + 1)
    ]

    if markers != expected_markers:
        raise ValueError(
            "ترتيب وسوم الصفحات غير صحيح"
        )

    return {
        "magic_value": True,
        "metadata_separator": True,
        "page_markers": len(markers),
        "page_order": True,
    }

# -*- coding: utf-8 -*-

"""
OpenITI Arabic OCR cleaning pipeline.

Input:
    volume_01/openiti_output/*.mARkdown

Output:
    volume_01/openiti_output/cleaned/
        *.mARkdown
        cleaning_changes.jsonl
        cleaning_report.json

Uses:
    openiti.helper.ara.normalize_ara_light

Preserves:
    - Original source file
    - OpenITI metadata
    - Page markers
    - Numeric information
    - Paragraph order
    - Empty pages
    - OCR errors (no guessing or correction)

Creates an audit record for every modified text line.
"""

import re

# ============================================================

OUTPUT_DIR = OPENITI_DIR / "cleaned"

# Arabic normalization options.
#
# These options follow OpenITI normalize_ara_light.
#
# denoise_text:
#     Remove tashkeel, shadda, tatweel and other noise.
#
# alif_seat:
#     أ إ آ ٱ -> ا
#
# alif_maqsura:
#     ى -> ي
#
# hamza:
#     Normalize hamza on waw/ya carriers.
#
# per_to_ara:
#     Convert Persian kaf/ya to Arabic kaf/ya.



import inspect

OPTIONS = {
    "denoise_text": True,
    "alif_seat": True,
    "alif_maqsura": True,
    "hamza": True,
    "per_to_ara": True,
}

# مرّر فقط الخيارات التي تدعمها الدالة فعلاً
_SUPPORTED = set(
    inspect.signature(ara.normalize_ara_light).parameters
)
ACTIVE_OPTIONS = {
    k: v for k, v in OPTIONS.items() if k in _SUPPORTED
}


# Preserve the original file.
OVERWRITE = False

# Include a maximum number of changed examples in report.
MAX_EXAMPLES = 100

def sha256_text(text):

    return hashlib.sha256(
        text.encode("utf-8")
    ).hexdigest()


def write_text(path, text):

    path.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    with path.open(
        "w",
        encoding="utf-8",
        newline=""
    ) as f:

        f.write(text)


def write_json(path, data):

    write_text(
        path,
        json.dumps(
            data,
            ensure_ascii=False,
            indent=2
        )
    )


def is_page_marker(line):

    return bool(
        re.fullmatch(
            r"Page(?:V\d+P\d+|Beg|End)",
            line.strip()
        )
    )


def is_metadata_line(line):

    return line.startswith("#META#")


def is_openiti_control_line(line):

    """
    Detect OpenITI structural lines that must not be normalized.
    """

    if not line:
        return True

    if line.startswith("######OpenITI#"):
        return True

    if is_metadata_line(line):
        return True

    if is_page_marker(line):
        return True

    # OpenITI structural tags.
    if re.match(
        r"^#{2,}\s*(?:\||\$|@|~|Page)",
        line
    ):
        return True

    return False


# ============================================================
# 3. CLEAN ONE TEXT LINE
# ============================================================


def clean_arabic_line(line):

    """
    Clean only the text content of an OpenITI paragraph.

    Input:
        # أَحْمَدُ بنُ يَحْيَى

    Output:
        # احمد بن يحيي

    The OpenITI paragraph marker is retained.
    """

    if not line.startswith("# "):
        return line

    prefix = "# "
    original_text = line[len(prefix):]
    

    cleaned_text = ara.normalize_ara_light(
        original_text,
        **ACTIVE_OPTIONS
    )
    

    return prefix + cleaned_text

# 4. CLEAN COMPLETE DOCUMENT
# ============================================================


def clean_document(original):

    lines = original.splitlines(keepends=True)

    cleaned_lines = []

    changes = []

    statistics = {
        "total_lines": len(lines),
        "text_lines": 0,
        "changed_lines": 0,
        "unchanged_lines": 0,
        "metadata_lines": 0,
        "page_markers": 0,
        "empty_lines": 0,
        "control_lines": 0,
        "removed_characters": 0,
    }

    for line_number, raw_line in enumerate(lines, start=1):

        # Preserve line ending.
        if raw_line.endswith("\r\n"):
            body = raw_line[:-2]
            ending = "\r\n"

        elif raw_line.endswith("\n"):
            body = raw_line[:-1]
            ending = "\n"

        elif raw_line.endswith("\r"):
            body = raw_line[:-1]
            ending = "\r"

        else:
            body = raw_line
            ending = ""

        if not body:
            statistics["empty_lines"] += 1
            cleaned_lines.append(raw_line)
            continue

        if is_metadata_line(body):
            statistics["metadata_lines"] += 1
            cleaned_lines.append(raw_line)
            continue

        if is_page_marker(body):
            statistics["page_markers"] += 1
            cleaned_lines.append(raw_line)
            continue

        if is_openiti_control_line(body):
            statistics["control_lines"] += 1
            cleaned_lines.append(raw_line)
            continue

        if body.startswith("# "):

            statistics["text_lines"] += 1

            cleaned_body = clean_arabic_line(body)

            if cleaned_body != body:

                statistics["changed_lines"] += 1

                statistics["removed_characters"] += (
                    len(body) - len(cleaned_body)
                )

                changes.append({
                    "line_number": line_number,
                    "original": body,
                    "cleaned": cleaned_body,
                    "original_length": len(body),
                    "cleaned_length": len(cleaned_body),
                })

            else:
                statistics["unchanged_lines"] += 1

            cleaned_lines.append(
                cleaned_body + ending
            )

        else:

            # Unknown non-paragraph lines are preserved.
            statistics["control_lines"] += 1
            cleaned_lines.append(raw_line)

    return "".join(cleaned_lines), changes, statistics

def validate_document(text):
